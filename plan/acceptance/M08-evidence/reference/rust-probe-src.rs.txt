use rust_skia::*;

fn show(name: &str, pe: Option<SkPathEffect>) {
    match pe {
        None => println!("{} null=1", name),
        Some(pe) => match pe.as_a_dash() {
            None => println!("{} valid=0", name),
            Some(info) => {
                println!(
                    "{} valid=1 n={} phase={:08x}",
                    name,
                    info.f_intervals.len(),
                    info.f_phase.to_bits()
                );
                for v in &info.f_intervals {
                    println!("  i {:08x}", v.to_bits());
                }
            }
        },
    }
}

#[test]
fn dev_as_a_dash_probe() {
    let iv = [4.0f32, 2.0, 1.0, 3.0];
    show("corner", SkCornerPathEffect::make(1.0));
    show("dash", SkDashPathEffect::make(&iv, 2.0));
    show("dashneg", SkDashPathEffect::make(&iv, -7.0));
    show("dashwrap", SkDashPathEffect::make(&iv, 27.0));
    show("dashnegzero", SkDashPathEffect::make(&iv, -10.0));
    show("dashtinyneg", SkDashPathEffect::make(&iv, -f32::from_bits(1)));
    show("odd", SkDashPathEffect::make(&[4.0, 2.0, 1.0], 0.0));
    show("zero", SkDashPathEffect::make(&[0.0, 0.0], 0.0));
    show("nan", SkDashPathEffect::make(&[f32::NAN, 1.0], 0.0));
    show("inf", SkDashPathEffect::make(&[f32::INFINITY, 1.0], 0.0));
    show("maxsum", SkDashPathEffect::make(&[f32::MAX, f32::MAX], 0.0));
    show("phnan", SkDashPathEffect::make(&iv, f32::NAN));
    show("phinf", SkDashPathEffect::make(&iv, f32::INFINITY));
    show("zeroton", SkDashPathEffect::make(&[0.0, 1.0], 0.0));
    show("corner0", SkCornerPathEffect::make(0.0));
    show("cornernan", SkCornerPathEffect::make(f32::NAN));
    show("cornerinf", SkCornerPathEffect::make(f32::INFINITY));
    show("cornertiny", SkCornerPathEffect::make(f32::from_bits(1)));
}
